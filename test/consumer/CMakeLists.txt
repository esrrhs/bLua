cmake_minimum_required(VERSION 3.15)
project(bLuaConsumer LANGUAGES CXX)
find_package(bLua 1.0 CONFIG REQUIRED)
get_target_property(blua_links bLua::bLua INTERFACE_LINK_LIBRARIES)
if(NOT "Lua::Lua" IN_LIST blua_links)
    message(FATAL_ERROR "Export must reference Lua::Lua, not build-machine dependency paths")
endif()
add_executable(consumer main.cpp second.cpp)
target_link_libraries(consumer PRIVATE bLua::bLua)
enable_testing()
add_test(NAME consumer_runs COMMAND consumer)
